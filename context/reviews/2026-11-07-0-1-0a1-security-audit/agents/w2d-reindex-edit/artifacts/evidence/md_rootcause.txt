"""Root-cause split for markdown_refs on one long line (research timing, not a test).

Same inline-parse shape twice: path-shaped code spans (each becomes a ref and folds the line)
versus code spans with a space inside (parsed identically, path_shaped refuses, no line_context).
"""
import time
from vfs import _native as n
def t(body):
    s = time.perf_counter(); refs = n.markdown_refs([body])[0]; return time.perf_counter() - s, len(refs)
for label, unit in (("path-shaped `a.md`", b"`a.md` "), ("not-a-path `a b` ", b"`a b` ")):
    prev = None
    for N in (8000, 16000):
        dt, r = t(unit * N)
        ratio = f"{dt/prev:.2f}x" if prev else "-"
        print(f"{label} N={N:6d} body={len(unit)*N:7d}B refs={r:6d} {dt*1e3:8.1f} ms (x2 ratio {ratio})")
        prev = dt
