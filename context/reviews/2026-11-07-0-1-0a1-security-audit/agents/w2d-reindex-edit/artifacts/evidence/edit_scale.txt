== harness: edit_scale.py
"""Doubling probe of the public edit verb on InMemoryStorage (research timing, not a test).

Measures edit wall time at n and 2n for three fuzzy-replacer shapes, plus the longest
event-loop stall a concurrent ticker sees and the latency of a concurrent read by another
principal on the same mount.
"""
import asyncio, sys, time
from vfs.base import AsyncVirtualFileSystem
from vfs.storage.backends.memory import InMemoryStorage
from vfs.authority import Authority, Principal
from vfs.storage.replace import EditOperation
from vfs.models import Observation
from vfs.paths import Path

alice = Authority.of(Principal("alice"))
bob = Authority.of(Principal("bob"))

async def ticker(stop, gaps):
    last = time.perf_counter()
    while not stop.is_set():
        await asyncio.sleep(0.005)
        now = time.perf_counter(); gaps.append(now - last); last = now

async def probe(fs, label, path, content, edits, targets=None):
    if targets is None:
        r = await fs.write(path=path, content=content, authority=alice)
        assert r.success, r
    stop = asyncio.Event(); gaps = []
    tick = asyncio.create_task(ticker(stop, gaps))
    await asyncio.sleep(0.02)
    async def other_read():
        s = time.perf_counter() + 0.01
        await asyncio.sleep(0.01)
        r = await fs.read("/bob.txt", authority=bob); return time.perf_counter() - s, r.success
    reader = asyncio.create_task(other_read())
    s = time.perf_counter()
    if targets is None:
        r = await fs.edit(path=path, edits=edits, authority=alice)
    else:
        r = await fs.edit(observations=targets, edits=edits, authority=alice)
    dt = time.perf_counter() - s
    rd, rok = await reader
    stop.set(); await tick
    err = str(r.errors[0].kind) + ": " + r.errors[0].message[:60].replace("\n", " ") if r.errors else ""
    print(f"{label:46s} content={len(content):7d}B edit={dt*1e3:8.1f} ms success={r.success} "
          f"maxloopgap={max(gaps)*1e3:8.1f} ms bob_read={rd*1e3:8.1f} ms ok={rok} | {err}")
    return dt

async def main():
    fs = AsyncVirtualFileSystem(storage=InMemoryStorage())
    assert (await fs.write(path="/bob.txt", content="bob's file", authority=bob)).success
    print("== block_anchor_replacer: L one-char lines, 3-line old whose last anchor never occurs (O(L^2))")
    prev = None
    for L in (2000, 4000, 8000):
        dt = await probe(fs, f"L={L}", f"/a{L}.txt", "a\n" * L, [EditOperation(old="a\nx\nb", new="z")])
        if prev: print(f"   ratio x2 -> {dt/prev:.2f}x")
        prev = dt
    print("== levenshtein: one candidate, middle line of M chars that differs (O(M^2) pure Python)")
    prev = None
    for M in (500, 1000, 2000):
        dt = await probe(fs, f"M={M}", f"/l{M}.txt", "a\n" + "x" * M + "\nb\n", [EditOperation(old="a\n" + "y" * M + "\nb", new="z")])
        if prev: print(f"   ratio x2 -> {dt/prev:.2f}x")
        prev = dt
    print("== line_trimmed_replacer: L lines all trimmed-matching, O(L) start_pos sum + context split per match (O(L^2))")
    prev = None
    for L in (2000, 4000, 8000):
        dt = await probe(fs, f"L={L}", f"/t{L}.txt", "a\n" * L, [EditOperation(old="  a", new="z")])
        if prev: print(f"   ratio x2 -> {dt/prev:.2f}x")
        prev = dt
    print("== target multiplier: the M=1000 levenshtein edit applied to K targets in one call (edit_rows continues past each failure)")
    for K in (1, 2, 4):
        paths = [f"/k{K}_{i}.txt" for i in range(K)]
        for p in paths:
            assert (await fs.write(path=p, content="a\n" + "x" * 1000 + "\nb\n", authority=alice)).success
        await probe(fs, f"K={K}", None, "a\n" + "x" * 1000 + "\nb\n", [EditOperation(old="a\n" + "y" * 1000 + "\nb", new="z")],
                    targets=[Observation(path=Path(p)) for p in paths])
    await fs.close()

asyncio.run(main())
