----- e2e.py
import sys, resource, asyncio, os
sys.path.insert(0, ".")
from vfs.models import Entry
from vfs.paths import Path
from vfs.storage.backends.database import DatabaseStorage
async def main():
    data = open(sys.argv[1], "rb").read()
    db = os.path.join(os.getcwd(), "e2e.sqlite")
    if os.path.exists(db): os.remove(db)
    st = DatabaseStorage(url=f"sqlite+aiosqlite:///{db}")
    w = await st.write(entries=[Entry(path=Path("/up/report.docx"), data=data)], parents=True)
    print("write_success", w.success)
    before = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    res = await st.reindex()
    after = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    stat = (await st.stat(path=Path("/up/report.docx"))).observations[0]
    print("reindex_success", res.success, "render_status", stat.render_status)
    print("stored_bytes", len(data), "maxrss_growth_MiB_during_reindex", (after - before) >> 20)
    await st.close() if hasattr(st, "close") else None
asyncio.run(main())
