"""End-to-end: what a native panic does to a host serving several callers.
Trigger is operator config grep_wall_seconds=inf (no lower-trust input reaches the Duration conversion)."""
import asyncio, threading, sys
from vfs import AsyncVirtualFileSystem, VirtualFileSystem
from vfs.storage.backends.memory import InMemoryStorage

async def main():
    fs = AsyncVirtualFileSystem(storage=InMemoryStorage(grep_wall_seconds=float("inf")))
    r = await fs.write(path="/a.txt", content="hello needle world\n")
    print("write ok:", r.ok if hasattr(r, "ok") else r)
    results = await asyncio.gather(
        fs.grep("needle", allow_scan=True),
        fs.read(path="/a.txt"),
        return_exceptions=True,
    )
    for name, res in zip(("grep", "read"), results):
        if isinstance(res, BaseException):
            print(f"{name}: raised {type(res).__module__}.{type(res).__name__}; Exception subclass={isinstance(res, Exception)}: {res}")
        else:
            print(f"{name}: Result errors={getattr(res, 'errors', None)!r}"[:200])
    try:
        await fs.grep("needle", allow_scan=True)
    except Exception as e:
        print("caught by `except Exception`:", type(e).__name__)
    except BaseException as e:
        print("NOT caught by `except Exception`; escaped as", type(e).__name__)
    after = await fs.read(path="/a.txt")
    print("read after panic still served:", getattr(after, "errors", None) == [] or after)
    print("threads alive:", sorted(t.name for t in threading.enumerate()))
    try:
        await fs.close()
    except BaseException as e:
        print("close:", type(e).__name__, e)

asyncio.run(main())
print("event loop completed normally")

with VirtualFileSystem(storage=InMemoryStorage(grep_wall_seconds=float("inf"))) as sfs:

print("process exits normally")
