"""Doubling probe of markdown_refs on N links in one line vs N links on N lines (research timing, not a test)."""
import time
from vfs import _native as n
def t(body):
    s = time.perf_counter(); refs = n.markdown_refs([body])[0]; return time.perf_counter() - s, len(refs)
prev = None
for N in (2000, 4000, 8000, 16000):
    one = (b"[a](b.md) " * N)
    many = (b"[a](b.md)\n" * N)
    t1, r1 = t(one); t2, r2 = t(many)
    ratio = f"{t1/prev:.2f}x" if prev else "-"
    print(f"N={N:6d} one-line {len(one):7d} B refs={r1:6d} {t1*1e3:9.1f} ms (x2 ratio {ratio}) | N lines refs={r2:6d} {t2*1e3:7.1f} ms")
    prev = t1
