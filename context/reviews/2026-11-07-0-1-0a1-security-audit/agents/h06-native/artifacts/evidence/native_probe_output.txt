from vfs import _native as n
rng = random.Random(7); nonws = 0; examples = []
for g in n.supported_grammars():
    for _ in range(6):
        body = bytes(rng.choice(b"{}()[]<>;:,.'\"`\\/*#-=+ \n\tabcxyz019\xc3\xa9") for _ in range(rng.randint(0, 600)))
        cs = rng.choice([0, 1, 7, 64, 4096])
        rows = n.chunk_spans([(body, g)], chunk_size=cs)[0]
        if not rows: continue
        covered = bytearray(len(body))
        for s, e, *_ in rows: covered[s:e] = b"\x01" * (e - s)
        lost = bytes(b for b, c in zip(body, covered) if not c)
        if lost.strip():
            nonws += 1
            if len(examples) < 3: examples.append((g, len(body), len(lost), lost[:40]))
print("bodies with non-whitespace bytes absent from every span:", nonws)
